cmake_minimum_required(VERSION 3.16)
project(nmmp_private_loader_tests C)
enable_testing()
add_library(envelope STATIC ../Envelope.c ../vendor/monocypher/monocypher.c)
target_include_directories(envelope PUBLIC ..)
target_link_libraries(envelope PUBLIC z)
add_executable(crypto_test crypto_test.c)
target_link_libraries(crypto_test PRIVATE envelope)
add_test(NAME crypto_envelope COMMAND crypto_test ${CMAKE_CURRENT_SOURCE_DIR}/envelope.golden)
add_library(loader STATIC ../Loader.c)
target_link_libraries(loader PUBLIC envelope dl)
add_executable(loader_test loader_test.c failure_hooks.c)
target_link_libraries(loader_test PRIVATE loader)
set(FAILURE_WRAPS "-Wl,--wrap=calloc,--wrap=mmap,--wrap=munmap,--wrap=mprotect,--wrap=dlopen,--wrap=dlclose,--wrap=dlsym,--wrap=dlvsym")
set_target_properties(loader_test PROPERTIES LINK_FLAGS "${FAILURE_WRAPS}")
add_test(NAME loader_content COMMAND loader_test ${CMAKE_CURRENT_SOURCE_DIR}/content.golden)
add_executable(once_test once_test.c ../Once.c)
target_include_directories(once_test PRIVATE ..)
if (NOT ANDROID)
    target_link_libraries(once_test PRIVATE pthread)
endif ()
add_test(NAME loader_once COMMAND once_test)
set(NMMP_VM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../nmmvm/nmmvm/src/main/cpp/vm" CACHE PATH "Shared native VM source")
set(NMMP_STAGE0_FIXTURE "" CACHE FILEPATH "Python generated stage0-programs.c")
if (NMMP_STAGE0_FIXTURE)
    add_executable(stage0_test stage0_test.c ../Stage0.c "${NMMP_VM_SOURCE}/NativeVm.c" "${NMMP_STAGE0_FIXTURE}")
    target_include_directories(stage0_test PRIVATE .. "${NMMP_VM_SOURCE}/include")
    target_link_libraries(stage0_test PRIVATE envelope)
    add_test(NAME stage0_recovery COMMAND stage0_test ${CMAKE_CURRENT_SOURCE_DIR}/envelope.golden)
endif ()
if (ANDROID)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_library(test_inner SHARED module.c)
    target_link_libraries(test_inner PRIVATE m)
    set_target_properties(test_inner PROPERTIES C_VISIBILITY_PRESET hidden
            LINK_FLAGS "-Wl,--pack-dyn-relocs=none,--hash-style=sysv,-Bsymbolic,--exclude-libs,ALL,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/../inner.exports")
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module.content"
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_module.py"
                    "$<TARGET_FILE:test_inner>" "${CMAKE_CURRENT_BINARY_DIR}/module.content"
                    "${CMAKE_SYSROOT}/usr/lib/aarch64-linux-android/26" "${CMAKE_READELF}"
            DEPENDS test_inner prepare_module.py ../pack.py VERBATIM)
    add_custom_target(module_content ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/module.content")
    add_executable(module_test module_test.c failure_hooks.c)
    target_link_libraries(module_test PRIVATE loader)
    set_target_properties(module_test PROPERTIES LINK_FLAGS "${FAILURE_WRAPS}")
    set(NMMP_GENERATED_INIT_TESTS "" CACHE PATH "Directory produced by GenerateInitFixtures.java")
    if (NMMP_GENERATED_INIT_TESTS)
        foreach (BRANCH bound unbound)
            add_executable(bootstrap_${BRANCH} bootstrap_test.c ../InnerBootstrap.c
                    "${NMMP_VM_SOURCE}/VmInit.c"
                    "${NMMP_GENERATED_INIT_TESTS}/${BRANCH}/jni_init.c")
            target_include_directories(bootstrap_${BRANCH} PRIVATE .. stubs "${NMMP_VM_SOURCE}/include")
            target_compile_definitions(bootstrap_${BRANCH} PRIVATE JNI_OnLoad=nmmp_inner_on_load)
            if (BRANCH STREQUAL "bound")
                target_compile_definitions(bootstrap_${BRANCH} PRIVATE NMMP_TEST_BOUND=1)
            else ()
                target_compile_definitions(bootstrap_${BRANCH} PRIVATE NMMP_TEST_BOUND=0)
            endif ()
        endforeach ()
    endif ()
endif ()
