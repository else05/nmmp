cmake_minimum_required(VERSION 3.22)
project(nmmp_semantic)
set(NMMP_CPP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../main/cpp" CACHE PATH "VM source tree")
add_subdirectory("${NMMP_CPP_ROOT}/vm" vm)
add_library(nmmp_semantic SHARED SemanticBridge.cpp)
target_include_directories(nmmp_semantic PRIVATE "${NMMP_CPP_ROOT}" "${NMMP_CPP_ROOT}/vm")
target_link_libraries(nmmp_semantic nmmvm log)
option(NMMP_TEST_DECODE_WIPE "Observe actual interpreter decoded-state exits (tests only)" OFF)
if (NMMP_TEST_DECODE_WIPE)
    target_sources(nmmvm PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/DecodedWipeBridge.cpp")
    target_include_directories(nmmvm PRIVATE "${NMMP_CPP_ROOT}/vm")
    target_compile_definitions(nmmvm PRIVATE NMMP_TEST_DECODE_WIPE=1)
endif ()
if (NMMP_BASELINE)
    target_compile_definitions(nmmp_semantic PRIVATE NMMP_BASELINE=1)
endif ()
if (NMMP_TEST_DEMAND)
    target_compile_definitions(nmmp_semantic PRIVATE NMMP_TEST_DEMAND=1)
endif ()
if (NMMP_TEST_NATIVE_VM)
    add_executable(nmmp_native_vm_test ../cpp/NativeVmTest.cpp "${NMMP_CPP_ROOT}/vm/NativeVm.c")
    target_include_directories(nmmp_native_vm_test PRIVATE "${NMMP_CPP_ROOT}/vm/include")
    add_executable(nmmp_native_root_test ../cpp/NativeRootTest.cpp "${NMMP_CPP_ROOT}/vm/VmCodec.cpp"
            "${NMMP_CPP_ROOT}/vm/NativeVm.c" "${NMMP_CPP_ROOT}/vm/VmInit.c")
    target_include_directories(nmmp_native_root_test PRIVATE "${NMMP_CPP_ROOT}/vm/include")
endif ()
if (NMMP_GENERATED_INIT)
    add_library(nmmp_init_semantic SHARED InitBridge.c "${NMMP_CPP_ROOT}/vm/VmInit.c")
    target_include_directories(nmmp_init_semantic PRIVATE "${NMMP_CPP_ROOT}/vm/include")
    target_compile_definitions(nmmp_init_semantic PRIVATE NMMP_GENERATED_INIT="${NMMP_GENERATED_INIT}")
endif ()
